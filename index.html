async function captureAndProcess() { const video = document.getElementById('video'); const canvas = document.getElementById('cropCanvas'); // 400x100 boyutunda olsun const ctx = canvas.getContext('2d'); // 1. Dinamik Kırpma: Videonun tam ortasından çerçeve boyutunda bir alan al const scaleX = video.videoWidth / video.clientWidth; const scaleY = video.videoHeight / video.clientHeight; // Çerçevenin ekrandaki konumunu (CSS üzerinden) dinamik hesapla const overlay = document.querySelector('.overlay'); const rect = overlay.getBoundingClientRect(); const videoRect = video.getBoundingClientRect(); const startX = (rect.left - videoRect.left) * scaleX; const startY = (rect.top - videoRect.top) * scaleY; const width = rect.width * scaleX; const height = rect.height * scaleY; // Canvas'a çiz (Bu aşamada canvas'ı display:block yapıp neyi kestiğini gör!) ctx.drawImage(video, startX, startY, width, height, 0, 0, 400, 100); // 2. Görüntü İyileştirme: Gri tonlama ve Renkleri Ters Çevir (Invert) let imageData = ctx.getImageData(0, 0, canvas.width, canvas.height); let data = imageData.data; for (let i = 0; i < data.length; i += 4) { let avg = (data[i] + data[i + 1] + data[i + 2]) / 3; // Invert: 255 - değer (Beyazı siyah, siyahı beyaz yap) let inverted = 255 - avg; data[i] = data[i+1] = data[i+2] = (inverted > 128) ? 255 : 0; // Eşikleme (Binarize) } ctx.putImageData(imageData, 0, 0); // 3. Tesseract Analizi const status = document.getElementById('result'); status.innerText = "Analiz ediliyor..."; const worker = await Tesseract.createWorker('eng'); await worker.setParameters({ tessedit_char_whitelist: '0123456789.', tessedit_pageseg_mode: '7' }); const { data: { text } } = await worker.recognize(canvas); status.innerText = "Sonuç: " + (text.trim() || "Okunamadı"); await worker.terminate(); }
